  add_compile_options(-fsigned-char)

# WWHD_BUNDLED_DEPS: build glslang, zlib, LZ4 (and SDL3 if not found) from pinned sources with the
# project's own compiler. Release builds use it so the prebuilt runtime and its dependencies all
# come from the toolchain the installer later links with (llvm-mingw, zig).
option(WWHD_BUNDLED_DEPS "build missing dependencies from pinned sources" OFF)
if(WIN32 AND CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC" OR WWHD_BUNDLED_DEPS)
  include(cmake/WindowsDependencies.cmake)
endif()
if(WIN32 AND CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC")
  add_compile_definitions(NOMINMAX _CRT_SECURE_NO_WARNINGS)
  # The MSVC ABI links no compiler runtime, but clang lowers 128-bit division to compiler-rt calls
  # (__udivti3; on ARM64 nothing in the MSVC libraries provides them): link clang's builtins.
  set(WWHD_CLANG_TARGET_ARG)
  if(CMAKE_CXX_COMPILER_TARGET)
    set(WWHD_CLANG_TARGET_ARG --target=${CMAKE_CXX_COMPILER_TARGET})
  endif()
  execute_process(COMMAND ${CMAKE_CXX_COMPILER} ${WWHD_CLANG_TARGET_ARG} -print-libgcc-file-name -rtlib=compiler-rt
                  OUTPUT_VARIABLE WWHD_CLANG_BUILTINS OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(WWHD_CLANG_BUILTINS AND EXISTS "${WWHD_CLANG_BUILTINS}")
    link_libraries("${WWHD_CLANG_BUILTINS}")
  endif()
endif()

# Every test runs with a user folder of its own (WWHD_USER_DIR: host::config_dir and all that follows it,
# controls.json, settings.ini, the mod manager and its story caches, and on macOS also the states,
# screenshots and display.plist, main.cpp apply_portable_mode), so no test writes into the player's
# folder (button_glyphs saved the Automatic face buttons to the real controls.json). A test that names
# its own WWHD_USER_DIR keeps it; one that sets it in code overrides this. Deferred to the end of this
# file so it reaches every test; user_folder_isolation checks it with HOME pointed elsewhere.
function(wwhd_isolate_test_user_folders)
  get_property(tests DIRECTORY PROPERTY TESTS)
  foreach(test IN LISTS tests)
    get_test_property(${test} ENVIRONMENT environment)
    if(NOT environment MATCHES "(^|;)WWHD_USER_DIR=")
      set_property(TEST ${test} APPEND PROPERTY ENVIRONMENT "WWHD_USER_DIR=${CMAKE_BINARY_DIR}/test-users/${test}")
    endif()
  endforeach()
endfunction()
cmake_language(DEFER CALL wwhd_isolate_test_user_folders)
